CREATE FULLTEXT INDEX lessonFullText
FOR (l:Lesson)
ON EACH [l.name, l.name_norm];

CALL db.index.fulltext.queryNodes(
  "lessonFullText",
  "python"
)
YIELD node, score
RETURN node.id, node.name, score
ORDER BY score DESC;
